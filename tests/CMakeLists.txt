macro(lmcas_add_test TARGET_NAME SOURCE_FILE)
    add_executable(${TARGET_NAME} ${SOURCE_FILE})
    target_link_libraries(${TARGET_NAME} PRIVATE lmcas)
    add_test(NAME ${TARGET_NAME} COMMAND ${TARGET_NAME})
    # Ensure test binaries go to the standard output directory
    set_target_properties(${TARGET_NAME} PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}"
    )
endmacro()

function(lmcas_add_public_header_check TARGET_NAME HEADER_NAME)
    add_library(${TARGET_NAME} OBJECT test_public_header_isolation.cpp)
    target_link_libraries(${TARGET_NAME} PRIVATE lmcas)
    target_compile_definitions(
        ${TARGET_NAME}
        PRIVATE LMCAS_HEADER_UNDER_TEST="${HEADER_NAME}"
    )
endfunction()
